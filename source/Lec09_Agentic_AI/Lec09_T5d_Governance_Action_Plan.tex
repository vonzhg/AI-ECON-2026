% ==== Phase B · T5-F12 · TIER: CORE (verified journal pointer)
\begin{frame}{Security: Red Lines and Workarounds}
\textbf{Do not put in front of an agent:} IRB-protected interviews, linked administrative microdata, personally identifiable information, HIPAA-regulated data, passwords / API keys / access tokens, embargoed or restricted-use datasets.

\vspace{0.2cm}

\textbf{Operational rule:} if you would not paste it into a public collaboration tool, do not paste it into an agent conversation.

\vspace{0.25cm}

\textbf{Sensitive data does not imply ``no agents ever.''} Three workarounds:
\begin{enumerate}
    \item Write the script with the agent, then run it yourself on the restricted data
    \item Use synthetic or masked data with the same structure for debugging
    \item Use institutionally approved local or private deployments when available
\end{enumerate}

\vspace{0.15cm}
\textbf{Safe pattern:} expose structure and summary statistics, not raw records.

\vspace{0.1cm}
{\scriptsize \textit{IRB, HIPAA, and GDPR rules apply unchanged to agent-aided work; journal disclosure rules are two frames on.}}
\end{frame}

% ==== Phase B · T1b-F14 · TIER: READ (pointer edits)
\begin{frame}{Treat Everything the Agent Reads as Untrusted}
\textbf{The defining agentic-AI security failure is prompt injection: data the agent merely \emph{reads} can hijack what it \emph{does}.}

\vspace{0.14cm}

\begin{itemize}\footnotesize
    \item \textbf{OWASP Top-10 for Agentic Applications} (Dec 2025) now codifies these risks.
    \item Documented exploits: \textbf{EchoLeak} (CVE-2025-32711, zero-click M365 Copilot data exfiltration); \textbf{ShadowPrompt} in Claude's Chrome extension; Anthropic's \textbf{GTG-1002} --- the first largely-autonomous cyberattack (Nov 2025).
    \item \textbf{NIST CAISI} red-teaming raised task-hijacking success from \textbf{11\% to 81\%} with adaptive attacks.
\end{itemize}

\vspace{0.12cm}
\begin{cautionbox}
\footnotesize\textbf{Operating rule.} Treat any content the agent reads --- emails, web pages, retrieved documents, even tool descriptions --- as an \emph{untrusted injection channel}. Use least-privilege sandboxing and revocable permissions, exactly as T5b's \texttt{settings.json} \texttt{deny} list and the previous frame prescribe. \textbf{[solid]}
\end{cautionbox}
\end{frame}

% ==== Phase B · NEW:Disclosure, Replication Packages, and \texttt{AI\_LOG.md} · TIER: READ
\begin{frame}{Disclosure, Replication Packages, and \texttt{AI\_LOG.md}}
\textbf{Journals already have rules for this. An agent-assisted paper has to meet them --- and the evidence has to exist before submission, not be reconstructed after.}

\vspace{0.15cm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\footnotesize\textbf{What the AEA journals require}
\begin{itemize}\scriptsize
  \item ``Artificial intelligence (AI) software, such as chatbots or other large language models, may not be listed as an author.''
  \item If AI software was used in preparing the manuscript, ``including drafting or editing text,'' it must be briefly described at submission; authors remain accountable for checking what it produced
  \item Papers are published ``only if the data and code used in the analysis are clearly and precisely documented'' --- agent-written code included
\end{itemize}
{\tiny\textit{AER submission guidelines and AEA journal policies, aeaweb.org, checked 12 Sep 2026. Other journals differ: read the one you submit to.}}
\end{column}
\begin{column}{0.47\textwidth}
\footnotesize\textbf{What that asks of your workflow}
\begin{itemize}\scriptsize
  \item \textbf{Replication package:} the code as run, the data or access instructions, the versions --- not the chat
  \item \textbf{\texttt{AI\_LOG.md}} (the capstone asks for one): the prompts that mattered, the decisions you made, where the AI was wrong and how you caught it, and an honest split of the work
  \item \textbf{Chain of custody:} date, harness and version, model --- next to every result you keep
\end{itemize}
\end{column}
\end{columns}

\vspace{0.12cm}
\begin{takeawaybox}
\footnotesize\textbf{Disclosure is cheap if you logged as you went, and impossible to fake afterwards.} A referee may ask which numbers an agent produced; your log is the answer.
\end{takeawaybox}
\end{frame}

% ==== Phase B · T1b-F15 · TIER: READ (ties to the seven tests)
\begin{frame}{Reproducibility: Chain-of-Custody for Agentic Results}
\textbf{Can an agent reproduce a published computational result? Often only partly --- which makes logging non-negotiable.}

\vspace{0.16cm}

\begin{itemize}\footnotesize
    \item \textbf{CORE-Bench}: agents reproduce published computational results only $\sim$\textbf{60\% (easy)} / $\sim$\textbf{21\% (hard)}. \textbf{[solid]}
    \item Therefore \textbf{log full trajectories}: prompts, tool calls, seeds, \emph{model versions}, \emph{spec versions}, and \emph{dates} --- the chain-of-custody for any agent-produced artifact.
\end{itemize}

\vspace{0.12cm}
\begin{cautionbox}
\footnotesize\textbf{Read vendor safety numbers skeptically.} Distinguish a vendor's ``$\sim$1\% attack success on a \emph{fixed} suite'' from academic findings that \emph{adaptive} attacks still exceed $\sim$85\%. A fixed-suite number measures the suite, not the threat model. \textbf{[solid]}
\end{cautionbox}

\vspace{0.1cm}
\footnotesize\textbf{Ties to the seven tests below.} This is the same discipline (reproducibility, verifiability, auditability) --- now with the dates and spec versions the landscape forces you to pin.
\end{frame}

\section{Measuring Your Own Agent}

\sectiondivider{Measuring Your Own Agent}{A headline number is not a measurement until you know its validity}

% ==== Phase B · T1b-F12 · TIER: READ
\begin{frame}{Benchmarks Break: What SWE-bench Verified Taught}
\textbf{The course's measurement theme, pointed at agents: a headline score is not a measurement until you know what it measures and who could have gamed it.}

\vspace{0.14cm}
\begin{itemize}\footnotesize
    \item \textbf{Contamination.} On 23 February 2026 OpenAI stopped reporting SWE-bench Verified: auditing 138 hard tasks, it found 59\% had flawed tests, and frontier models could reproduce gold patches verbatim. It recommends SWE-bench Pro instead. \textbf{[solid]}
    \item \textbf{Harness exploits.} In April 2026 Berkeley RDI scored near 100\% on eight agent benchmarks --- SWE-bench Verified and Pro, Terminal-Bench, WebArena among them --- \emph{without solving a single task}, mostly without calling a model. \textbf{[solid]}
    \item \textbf{Never compare across suites.} A Verified score and a Pro score are different problems, difficulty, and exposure --- never one axis.
\end{itemize}

\vspace{0.1cm}
\begin{cautionbox}
\footnotesize\textbf{Your own evaluations are benchmarks too.} Berkeley's rule transfers directly to the Design Lab: run a do-nothing agent against your detection matrix. If it scores above zero, your test has a bug. (Reliability over repeated runs: T4.)
\end{cautionbox}

\vspace{0.04cm}
{\scriptsize\textit{OpenAI, ``Why SWE-bench Verified no longer measures frontier coding capabilities'' (Feb 2026); Berkeley RDI, ``How We Broke Top AI Agent Benchmarks'' (Apr 2026). Both checked 12 Sep 2026.}}
\end{frame}

\sectiondivider{Limits and Judgment}{What the workflow changes, and what remains human}

% ==== Phase B · T1-F13..T6-F06 · TIER: CORE
\begin{frame}{Seven Tests for Any AI-Assisted Research Output}
\textbf{Before a script, figure, or table an agent produced goes into your paper, put it through seven questions. Any ``no'' or ``I don't know'' means it is not yet research-grade.}

\vspace{0.12cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.24}
\begin{tabular}{|>{\raggedright\arraybackslash}p{2.9cm}|>{\raggedright\arraybackslash}p{9.4cm}|}
\hline
\textbf{Test} & \textbf{The question} \\ \hline
1. Correctness & Did it compute, extract, or summarize the right object --- and can you re-check the math, data flow, and magnitudes independently? \\ \hline
2. Reproducibility & Can a co-author rerun it from a clean clone --- pinned model, prompt, seed, and data version? \\ \hline
3. Auditability & Can a referee trace the commands, checks, and sources that produced it? \\ \hline
4. Privacy boundary & Which files stayed local, and what content may still have crossed the network? \\ \hline
5. Cost & What did it cost --- in tokens and dollars, including what a replicator will pay? \\ \hline
6. Speed & Net of checking and fixing, did it actually save time? \\ \hline
7. Skill investment & Is your team building durable skills --- or quietly losing them? \\ \hline
\end{tabular}
\end{center}

\vspace{0.1cm}
\footnotesize\textit{Distinct from T5a's five plan-grading criteria, which judge a plan before any code exists; these judge the output.}
\end{frame}

% ==== Phase B · T6-F07 · TIER: READ
\begin{frame}{Three Decisions You Cannot Delegate}
\textbf{Not because agents cannot talk about them --- they can, fluently. Because the researcher bears the consequences, and a delegated decision still has your name on it.}

\vspace{0.12cm}
\begin{itemize}
  \item \textbf{Identification --- which variation answers the causal question.} \\
  {\small An agent can describe instruments, event windows, and shocks, and sometimes spot a weak one. It cannot \emph{certify} that a shock is exogenous in your setting; that claim is yours to defend in a seminar.}

  \vspace{0.08cm}
  \item \textbf{Welfare --- which objective defines ``good.''} \\
  {\small Pareto weights, discount rates, the equity--efficiency trade-off: normative choices, not technical ones.}

  \vspace{0.08cm}
  \item \textbf{Judgment --- which question is worth asking.} \\
  {\small An agent optimizes the target it is given; deciding which target is interesting, novel, and policy-relevant is the job.}
\end{itemize}

\vspace{0.1cm}
\begin{takeawaybox}
\footnotesize\textbf{Capability is not the issue.} HA-Ramsey's AI iteration surfaced real methodological discoveries (Lec10 T8), and Lec10 T9's assistant passed every gate for the wrong reason. In both, the economist decided what counted. That is the residual claim a principal keeps.
\end{takeawaybox}
\end{frame}

% ==== Phase B · T6-F08 · TIER: READ (red-lines pointer; seven tests; list tightened)
\begin{frame}{When NOT to Use an Agent}
\textbf{The seven tests also mark a boundary: some research tasks cost more to delegate than they return, or fail too expensively.}

\vspace{0.1cm}

\begin{itemize}\small
    \item \textbf{One-line scripts and isolated edits.} A two-line \texttt{sed} or one-shot plot has no validation surface for the harness to add value. Open the editor.
    \item \textbf{Deeply proprietary or undocumented pipelines.} If the conventions live in one person's head, the agent has nothing to read; the time you would spend documenting the pipeline well enough to brief the agent is the same time you would spend just fixing the thing.
    \item \textbf{Strict deterministic guarantees.} For numerical reproducibility audits, regulatory pipelines, or replication packages, model nondeterminism is a hazard the harness cannot fully suppress. Run it pinned; do not let the agent retry.
    \item \textbf{Restricted-data analyses where prompts may leak.} If pasting one column would violate IRB or HIPAA, the agent does not belong in the loop; the red-lines workarounds apply.
    \item \textbf{Tasks where you do not yet have a benchmark.} Without a way to validate, faster output just means faster wrong. Define the V0 benchmark first; \emph{then} bring in the agent.
\end{itemize}

\vspace{0.1cm}
\footnotesize\textbf{Practical rule:} the agent is most useful when the work is bounded, validated, and repeatable. None of those? Skip the agent.
\end{frame}

% ---- T6-F09 · KEEP · TIER: READ · from T6:197
\begin{frame}{Trust but Verify}
\begin{enumerate}
    \item Reproduce one statistic by hand
    \item Cross-check one result with another package or script
    \item Test a limiting case or zero-variance benchmark
    \item Read the code line by line where the economics enters
    \item Ask whether the magnitude and sign make sense
\end{enumerate}

\vspace{0.3cm}

\textbf{This is not optional ceremony.} It is what turns fast code into defensible research.
\end{frame}

% ---- T6-F10 · KEEP · TIER: READ · from T6:211
\begin{frame}{What Changes for Economic Research}
\begin{itemize}
    \item The distance from question to first result shrinks
    \item More specifications become cheap enough to try
    \item Reproducibility can improve because workflow is written down
    \item The premium shifts away from typing speed and toward design, judgment, and validation
\end{itemize}

\vspace{0.3cm}

\textbf{What stays human:} asking the question, choosing the design, knowing the data, and interpreting the output.
\end{frame}

\begin{frame}[fragile]{Your First Exercise: Clean Up Old Code}
\textbf{The best first exercise is not ``do research for me.'' It is: clean up one of your own old projects --- safely, on a branch, and prove nothing changed.}

\vspace{0.1cm}
\begin{lstlisting}[style=pythonstyle,language={},basicstyle=\ttfamily\tiny,numbers=none]
I want your help cleaning up this project and setting up a git repo.
First, copy it all over to a backup directory.
Next, suggest an initial gitignore and create a first repo.
Then, push the initial repo to a private repo on GitHub.
Then, check out a new branch and suggest how to restructure the repo
  to have best practices with data/code/docs/paper organization.
Then just do it on the new branch.
Finally, actually run any code you've modified/restructured to make
  sure it lines up with the prior results.
\end{lstlisting}

\vspace{0.05cm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\footnotesize\textbf{Why each line is there}
\begin{itemize}\scriptsize
    \item \textbf{backup directory:} you can always go back
    \item \textbf{gitignore:} code apart from data and output
    \item \textbf{private repository:} history off your machine
\end{itemize}
\end{column}
\begin{column}{0.47\textwidth}
\begin{itemize}\scriptsize
    \item \textbf{new branch:} the restructuring is reversible
    \item \textbf{rerun and compare:} the check that the clean-up changed the layout, not the results
\end{itemize}
\end{column}
\end{columns}

\vspace{0.06cm}
{\tiny\textit{Prompt adapted from Aniket Panjwani, ``AI Agents for Economics Research'' (CEPR/VoxDev webinar, 2026). Private data stays out of the repository: T5d's red lines apply.}}
\end{frame}
